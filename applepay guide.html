Route 1 — Create Stripe PaymentIntent

app/api/stripe/create-intent/route.ts
import { NextRequest, NextResponse } from 'next/server'

export async function POST(req: NextRequest) {
  try {
    // User ka auth token header se lo
    const authToken = req.headers.get('authorization')
    if (!authToken) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { amount } = await req.json()

    // Biogance backend call — iOS ki tarah bilkul same
    const res = await fetch(
      `${process.env.BIOGANCE_API_BASE}/user/payment/create/intent`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': authToken,
        },
        body: JSON.stringify({
          amount: String(amount),
          payment_method: 'card',  // Always "card" — iOS bhi yahi bhejta hai
        }),
      }
    )

    const data = await res.json()

    if (!res.ok || !data?.data?.client_secret) {
      return NextResponse.json(
        { error: data?.action ?? 'Failed to create payment intent' },
        { status: 400 }
      )
    }

    // client_secret → Stripe ke liye, paymentIntentId → place order ke liye
    return NextResponse.json({
      clientSecret:    data.data.client_secret,
      paymentIntentId: data.data.payment_intent_id,
    })

  } catch (err) {
    console.error('[create-intent]', err)
    return NextResponse.json({ error: 'Server error' }, { status: 500 })
  }
}

Route 2 — Place Order

app/api/orders/place/route.ts
import { NextRequest, NextResponse } from 'next/server'

export async function POST(req: NextRequest) {
  try {
    const authToken = req.headers.get('authorization')
    if (!authToken) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const body = await req.json()

    const res = await fetch(
      `${process.env.BIOGANCE_API_BASE}/user/order/place`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': authToken,
        },
        body: JSON.stringify(body),
      }
    )

    const data = await res.json()
    if (!res.ok) {
      return NextResponse.json({ error: 'Failed to place order' }, { status: 400 })
    }

    return NextResponse.json({ success: true, data: data.data })

  } catch (err) {
    console.error('[place-order]', err)
    return NextResponse.json({ error: 'Server error' }, { status: 500 })
  }
}

Step 2 — Apple Pay Button Component (Complete)

components/ApplePayButton.tsx
'use client'

import { useEffect, useRef, useState } from 'react'
import { useStripe } from '@stripe/react-stripe-js'

// ── Types ──────────────────────────────────────────────────────────
interface OrderForm {
  fullName:                string
  email:                   string
  countryCode:             string   // e.g. "+49"
  phone:                   string
  deliveryAddressType:     string
  deliveryAddressFull:     string
  deliveryCountry:         string
  deliveryCity:            string
  deliveryPostalCode:      string
  invoiceAddressType:      string
  invoiceAddressFull:      string
  invoiceCountry:          string
  invoiceCity:             string
  invoicePostalCode:       string
  isInvoiceSameAsDelivery: boolean
  shippingCost:            string
  subtotal:                string
  taxAmount:               string
  totalAmount:             string   // e.g. "49.99"
  voucher:                 string
  promoCode:               string
  cartIds:                 string   // e.g. "1,2,3"
}

interface Props {
  form:      OrderForm
  authToken: string
  onSuccess: (orderData: any) => void
  onError:   (msg: string)    => void
}

// ── Component ──────────────────────────────────────────────────────
export default function ApplePayButton({ form, authToken, onSuccess, onError }: Props) {

  const stripe           = useStripe()
  const [canPay, setCanPay]   = useState(false)
  const [loading, setLoading] = useState(false)
  const prRef = useRef<any>(null)

  // ── Check Apple Pay availability ────────────────────────────────
  useEffect(() => {
    if (!stripe) return

    const amountInCents = Math.round(parseFloat(form.totalAmount) * 100)

    const pr = stripe.paymentRequest({
      country:  'FR',          // Same as iOS
      currency: 'eur',         // Same as iOS (lowercase for Stripe web)
      total: {
        label:  'Biogance',    // Same as iOS PKPaymentSummaryItem label
        amount: amountInCents, // Stripe uses cents — 49.99 EUR = 4999
      },
      requestPayerName:  false,
      requestPayerEmail: false,
    })

    pr.canMakePayment().then((result) => {
      if (result?.applePay) {
        prRef.current = pr
        setCanPay(true)
      }
    })
  }, [stripe, form.totalAmount])

  // ── Handle click ────────────────────────────────────────────────
  const handleClick = async () => {
    if (!stripe || !prRef.current) return
    setLoading(true)

    let paymentIntentId = ''

    // ── paymentmethod event (fired after user approves in Apple Pay sheet) ──
    prRef.current.on('paymentmethod', async (event: any) => {
      try {

        // STEP 1 — Create Stripe PaymentIntent via Biogance backend ──────
        const intentRes = await fetch('/api/stripe/create-intent', {
          method: 'POST',
          headers: {
            'Content-Type':  'application/json',
            'Authorization': `Bearer ${authToken}`,
          },
          body: JSON.stringify({ amount: form.totalAmount }),
        })

        const intentData = await intentRes.json()

        if (!intentRes.ok || !intentData.clientSecret) {
          event.complete('fail')
          onError('Payment setup failed. Please try again.')
          setLoading(false)
          return
        }

        paymentIntentId = intentData.paymentIntentId

        // STEP 2 — Confirm PaymentIntent with Apple Pay token ────────────
        // Web equivalent of iOS: completion(clientSecret, nil)
        const { error: stripeError } = await stripe.confirmCardPayment(
          intentData.clientSecret,
          { payment_method: event.paymentMethod.id },
          { handleActions: false }
        )

        if (stripeError) {
          event.complete('fail')
          onError(stripeError.message ?? 'Payment failed')
          setLoading(false)
          return
        }

        // Apple Pay sheet pe checkmark show karo
        event.complete('success')

        // STEP 3 — Place Order ─────────────────────────────────────────
        // Same as iOS: orderPlaceAPI(paymentId: successApplePaymentIntentID)
        const placeRes = await fetch('/api/orders/place', {
          method: 'POST',
          headers: {
            'Content-Type':  'application/json',
            'Authorization': `Bearer ${authToken}`,
          },
          body: JSON.stringify({
            full_name:                    form.fullName,
            email:                        form.email,
            country_code:                 form.countryCode,
            phone:                        form.phone,
            phone_number:                 form.countryCode + form.phone,
            delivery_address_type:        form.deliveryAddressType,
            delivery_address_full:        form.deliveryAddressFull,
            delivery_address_country:     form.deliveryCountry,
            delivery_address_city:        form.deliveryCity,
            delivery_address_postal_code: form.deliveryPostalCode,
            invoice_address_type:         form.invoiceAddressType,
            invoice_address_full:         form.invoiceAddressFull,
            invoice_address_country:      form.invoiceCountry,
            invoice_address_city:         form.invoiceCity,
            invoice_address_postal_code:  form.invoicePostalCode,
            is_invoice_same_as_delivery:  form.isInvoiceSameAsDelivery,
            payment_status:  'Confirmed',  // Always "Confirmed"
            payment_method:  'Apple Pay',  // EXACT string — matches iOS
            shippingCost:    form.shippingCost,
            subtotal:        form.subtotal,
            taxAmount:       form.taxAmount,
            totalAmount:     form.totalAmount,
            payment_id:      paymentIntentId, // payment_intent_id — matches iOS
            voucher:         form.voucher,
            promo_code:      form.promoCode,
            cartIds:         form.cartIds,
          }),
        })

        const placeData = await placeRes.json()
        setLoading(false)

        if (!placeRes.ok) {
          onError(placeData.error ?? 'Order failed after payment')
          return
        }

        // SUCCESS ✓
        onSuccess(placeData.data)

      } catch (err: any) {
        event.complete('fail')
        setLoading(false)
        onError(err.message ?? 'Something went wrong')
      }
    })

    // Open Apple Pay sheet
    prRef.current.show()
  }

  // Don't render anything if Apple Pay not available
  if (!canPay) return null

  return (
    <button
      onClick={handleClick}
      disabled={loading}
      style={{
        width:          '100%',
        background:     '#000',
        color:          '#fff',
        border:         'none',
        borderRadius:   '8px',
        padding:        '14px 24px',
        fontSize:       '16px',
        fontWeight:     '600',
        cursor:         loading ? 'not-allowed' : 'pointer',
        opacity:        loading ? 0.6 : 1,
        fontFamily:     '-apple-system, BlinkMacSystemFont, sans-serif',
        display:        'flex',
        alignItems:     'center',
        justifyContent: 'center',
        gap:            '8px',
      }}
    >
      {loading ? 'Processing...' : ' Pay'}
    </button>
  )
}

9 Error Handling

ScenarioiOS BehaviorWeb Action
Apple Pay not available Shows alert Button auto-hidden (canMakePayment() returns null)
User cancels sheet Shows "User cancelled" alert Sheet closes silently — no action needed
create-intent API fails completion(nil, error) event.complete('fail') + show error banner
Stripe confirm fails Stripe SDK handles event.complete('fail') + show stripe error message
Place order fails (after payment charged) Silent in iOS ⚠️ Log paymentIntentId before calling — show support message
🚨
Critical scenario: Agar payment ho gaya (Stripe confirm success) lekin place order fail ho gaya — customer ka paise gaye order nahi aaya. Server-side paymentIntentId log karo place order call se pehle taake manually recover kar sako.
